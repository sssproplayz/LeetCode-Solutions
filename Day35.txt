class Solution {
public:
    bool checkValidString(string s) {
        int low = 0;
        int high = 0;

        for (char c : s) {
            if (c == '(') {
                low++;
                high++;
            }
            else if (c == ')') {
                low--;
                high--;
            }
            else { // '*'
                // '*' can be '(' or ')' or empty
                low--;
                high++;
            }

            // Even the maximum possible opens became negative
            if (high < 0)
                return false;

            // We cannot have fewer than 0 open brackets
            low = max(low, 0);
        }

        // Valid if zero open brackets is possible
        return low == 0;
    }
};
