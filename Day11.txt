class Solution {
public:

    // Prime factors of each digit.
    // Index = digit
    // Factors are stored as {2, 3, 5, 7}
    vector<vector<int>> factor = {
        {},          // 0
        {},          // 1
        {2},         // 2
        {3},         // 3
        {2, 2},      // 4
        {5},         // 5
        {2, 3},      // 6
        {7},         // 7
        {2, 2, 2},   // 8
        {3, 3}       // 9
    };

    vector<int> getPrimeCount(long long t) {
        vector<int> cnt(4, 0);

        int primes[4] = {2, 3, 5, 7};

        for (int i = 0; i < 4; i++) {
            while (t % primes[i] == 0) {
                t /= primes[i];
                cnt[i]++;
            }
        }

        // Remaining factor means t contains
        // a prime other than 2,3,5,7.
        if (t != 1)
            return {-1};

        return cnt;
    }

    vector<int> getPrimeCount(const string& s) {
        vector<int> cnt(4, 0);

        int primeValue[4] = {2, 3, 5, 7};

        for (char ch : s) {
            int d = ch - '0';

            for (int x : factor[d]) {
                for (int i = 0; i < 4; i++) {
                    if (primeValue[i] == x) {
                        cnt[i]++;
                        break;
                    }
                }
            }
        }

        return cnt;
    }

    vector<int> subtractFactors(
        vector<int> a,
        const vector<int>& b
    ) {
        for (int i = 0; i < 4; i++) {
            a[i] = max(0, a[i] - b[i]);
        }

        return a;
    }

    // Convert required prime factors into digits
    // using the minimum number of digits.
    vector<int> getFactorCount(vector<int> cnt) {

        // cnt[0] = number of 2s
        // cnt[1] = number of 3s
        // cnt[2] = number of 5s
        // cnt[3] = number of 7s

        int count8 = cnt[0] / 3;
        int remaining2 = cnt[0] % 3;

        int count9 = cnt[1] / 2;
        int count3 = cnt[1] % 2;

        int count4 = remaining2 / 2;
        int count2 = remaining2 % 2;

        int count6 = 0;

        // 2 + 3 -> 6
        if (count2 == 1 && count3 == 1) {
            count2 = 0;
            count3 = 0;
            count6 = 1;
        }

        // 4 + 3 -> 2 + 6
        if (count3 == 1 && count4 == 1) {
            count2 = 1;
            count6 = 1;
            count3 = 0;
            count4 = 0;
        }

        return {
            count2,
            count3,
            count4,
            cnt[2],
            count6,
            cnt[3],
            count8,
            count9
        };
    }

    int sumDigits(const vector<int>& cnt) {
        int ans = 0;

        for (int x : cnt)
            ans += x;

        return ans;
    }

    string construct(const vector<int>& cnt) {

        string ans;

        // Digits must be in ascending order.
        for (int digit = 2; digit <= 9; digit++) {
            ans += string(cnt[digit - 2], char('0' + digit));
        }

        return ans;
    }

    bool isSubset(
        const vector<int>& need,
        const vector<int>& have
    ) {
        for (int i = 0; i < 4; i++) {
            if (have[i] < need[i])
                return false;
        }

        return true;
    }

    string smallestNumber(string num, long long t) {

        // ----------------------------------------
        // 1. Factorize t
        // ----------------------------------------

        vector<int> primeCount = getPrimeCount(t);

        if (primeCount.size() == 1 &&
            primeCount[0] == -1) {
            return "-1";
        }


        // ----------------------------------------
        // 2. Get minimum digit representation
        // ----------------------------------------

        vector<int> factorCount =
            getFactorCount(primeCount);

        if (sumDigits(factorCount) > (int)num.size()) {

            return construct(factorCount);
        }


        // ----------------------------------------
        // 3. Count factors in num
        // ----------------------------------------

        vector<int> prefix =
            getPrimeCount(num);


        // ----------------------------------------
        // 4. Find first zero
        // ----------------------------------------

        int firstZero = num.find('0');

        if (firstZero == (int)string::npos) {

            firstZero = num.size();

            // num itself works.
            if (isSubset(primeCount, prefix)) {
                return num;
            }
        }


        // ----------------------------------------
        // 5. Work backwards
        // ----------------------------------------

        for (int i = (int)num.size() - 1;
             i >= 0;
             i--) {

            int d = num[i] - '0';


            // Remove current digit's factors
            // from the suffix calculation.
            vector<int> currentFactors(4, 0);

            for (int x : factor[d]) {

                if (x == 2)
                    currentFactors[0]++;

                else if (x == 3)
                    currentFactors[1]++;

                else if (x == 5)
                    currentFactors[2]++;

                else if (x == 7)
                    currentFactors[3]++;
            }

            prefix =
                subtractFactors(prefix,
                                currentFactors);


            int space =
                (int)num.size() - 1 - i;


            // We cannot keep a prefix containing
            // a zero.
            if (i > firstZero)
                continue;


            // ------------------------------------
            // Try a larger digit here
            // ------------------------------------

            for (int bigger = d + 1;
                 bigger <= 9;
                 bigger++) {

                vector<int> biggerFactors(4, 0);

                for (int x : factor[bigger]) {

                    if (x == 2)
                        biggerFactors[0]++;

                    else if (x == 3)
                        biggerFactors[1]++;

                    else if (x == 5)
                        biggerFactors[2]++;

                    else if (x == 7)
                        biggerFactors[3]++;
                }


                // Factors still required.
                vector<int> required =
                    subtractFactors(
                        primeCount,
                        prefix
                    );

                required =
                    subtractFactors(
                        required,
                        biggerFactors
                    );


                // Convert required factors into
                // minimum number of digits.
                vector<int> requiredDigits =
                    getFactorCount(required);


                int needed =
                    sumDigits(requiredDigits);


                if (needed <= space) {

                    int ones =
                        space - needed;


                    string ans;

                    // Original prefix.
                    ans += num.substr(0, i);

                    // Larger digit.
                    ans += char('0' + bigger);

                    // Fill unused positions with 1.
                    ans += string(ones, '1');

                    // Required factor digits.
                    ans += construct(requiredDigits);

                    return ans;
                }
            }
        }


        // ----------------------------------------
        // 6. No same-length answer.
        // Need one additional digit.
        // ----------------------------------------

        vector<int> required =
            getFactorCount(primeCount);

        int needed =
            sumDigits(required);

        string ans;

        ans += string(
            num.size() + 1 - needed,
            '1'
        );

        ans += construct(required);

        return ans;
    }
};
