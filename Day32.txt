class Solution {
public:
    int distinctEchoSubstrings(string text) {
        int n = text.size();

        const long long BASE = 31;
        const long long MOD = 1000000007LL;

        vector<long long> power(n + 1, 1);
        vector<long long> prefix(n + 1, 0);

        for (int i = 0; i < n; i++) {
            power[i + 1] = power[i] * BASE % MOD;
            prefix[i + 1] =
                (prefix[i] * BASE + (text[i] - 'a' + 1)) % MOD;
        }

        auto getHash = [&](int l, int r) {
            long long h =
                (prefix[r] - prefix[l] * power[r - l]) % MOD;

            if (h < 0)
                h += MOD;

            return h;
        };

        unordered_set<long long> seen;

        for (int len = 1; len * 2 <= n; len++) {
            for (int i = 0; i + 2 * len <= n; i++) {

                // First half
                long long h1 = getHash(i, i + len);

                // Second half
                long long h2 = getHash(i + len, i + 2 * len);

                if (h1 == h2) {
                    // Store the whole XX substring
                    long long fullHash = getHash(i, i + 2 * len);
                    seen.insert(fullHash);
                }
            }
        }

        return seen.size();
    }
};
