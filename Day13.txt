#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    int numDistinct(string s, string t) {

        int n = s.size();
        int m = t.size();

        const int LIMIT = INT_MAX;

        vector<vector<int>> dp(
            n + 1,
            vector<int>(m + 1, 0)
        );

        // Empty t can be formed in exactly 1 way
        dp[0][0] = 1;

        // Empty t can always be formed
        for (int i = 1; i <= n; i++) {
            dp[i][0] = 1;
        }

        for (int i = 1; i <= n; i++) {

            for (int j = 1; j <= m; j++) {

                // Don't use s[i - 1]
                dp[i][j] = dp[i - 1][j];

                // Use s[i - 1] if characters match
                if (s[i - 1] == t[j - 1]) {

                    long long value =
                        (long long)dp[i][j] +
                        dp[i - 1][j - 1];

                    // Prevent integer overflow
                    if (value > LIMIT) {
                        dp[i][j] = LIMIT;
                    } else {
                        dp[i][j] = (int)value;
                    }
                }
            }
        }

        return dp[n][m];
    }
};
