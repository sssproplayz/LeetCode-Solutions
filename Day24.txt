class Solution {
public:
    string longestDupSubstring(string s) {
        int n = s.size();

        const long long BASE = 31;
        const long long MOD = 1000000007LL;

        vector<long long> power(n + 1, 1);
        vector<long long> prefix(n + 1, 0);

        for (int i = 0; i < n; i++) {
            power[i + 1] = (power[i] * BASE) % MOD;
            prefix[i + 1] =
                (prefix[i] * BASE + (s[i] - 'a' + 1)) % MOD;
        }

        auto getHash = [&](int l, int r) {
            long long h =
                (prefix[r] - prefix[l] * power[r - l]) % MOD;

            if (h < 0)
                h += MOD;

            return h;
        };

        auto check = [&](int len) -> int {
            if (len == 0)
                return 0;

            unordered_map<long long, vector<int>> seen;

            for (int i = 0; i + len <= n; i++) {
                long long h = getHash(i, i + len);

                if (seen.count(h)) {
                    for (int start : seen[h]) {
                        if (s.compare(start, len, s, i, len) == 0)
                            return i;
                    }
                }

                seen[h].push_back(i);
            }

            return -1;
        };

        int low = 1;
        int high = n - 1;
        int bestStart = -1;
        int bestLen = 0;

        while (low <= high) {
            int mid = low + (high - low) / 2;

            int start = check(mid);

            if (start != -1) {
                bestLen = mid;
                bestStart = start;
                low = mid + 1;
            } else {
                high = mid - 1;
            }
        }

        if (bestStart == -1)
            return "";

        return s.substr(bestStart, bestLen);
    }
};
