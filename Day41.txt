class Solution {
public:

    bool isValid(string s) {
        int balance = 0;

        for (char c : s) {

            if (c == '(') {
                balance++;
            }
            else if (c == ')') {
                balance--;
            }

            if (balance < 0) {
                return false;
            }
        }

        return balance == 0;
    }

    vector<string> removeInvalidParentheses(string s) {

        vector<string> ans;

        queue<string> q;
        unordered_set<string> visited;

        q.push(s);
        visited.insert(s);

        bool found = false;

        while (!q.empty()) {

            int size = q.size();

            // Process one complete BFS level
            for (int i = 0; i < size; i++) {

                string current = q.front();
                q.pop();

                // Check whether current string is valid
                if (isValid(current)) {
                    ans.push_back(current);
                    found = true;
                }

                // If a valid answer was found,
                // don't generate strings with more removals.
                if (found) {
                    continue;
                }

                // Try removing every parenthesis
                for (int j = 0; j < current.length(); j++) {

                    if (current[j] != '(' && current[j] != ')') {
                        continue;
                    }

                    string next =
                        current.substr(0, j) +
                        current.substr(j + 1);

                    // Avoid duplicate strings
                    if (visited.find(next) == visited.end()) {
                        visited.insert(next);
                        q.push(next);
                    }
                }
            }

            // We found valid strings at this level.
            // Therefore, minimum removals are achieved.
            if (found) {
                break;
            }
        }

        return ans;
    }
};
