
class Solution {
public:
    long long minSumSquareDiff(vector<int>& nums1,
                               vector<int>& nums2,
                               int k1, int k2) {
        int n = nums1.size();
        vector<int> diff(n);

        long long total = 0;
        int mx = 0;
        long long k = (long long)k1 + k2;

        for (int i = 0; i < n; i++) {
            diff[i] = abs(nums1[i] - nums2[i]);
            total += diff[i];
            mx = max(mx, diff[i]);
        }

        if (total <= k)
            return 0;

        int left = 0, right = mx;

        while (left < right) {
            int mid = left + (right - left) / 2;
            long long need = 0;

            for (int d : diff) {
                need += max(0, d - mid);
            }

            if (need <= k)
                right = mid;
            else
                left = mid + 1;
        }

        long long ans = 0;

        for (int d : diff) {
            int reduced = min(d, left);
            k -= max(0, d - left);
            ans += 1LL * reduced * reduced;
        }

        for (int i = 0; i < n && k > 0; i++) {
            if (diff[i] >= left && left > 0) {
                ans -= 1LL * left * left;
                ans += 1LL * (left - 1) * (left - 1);
                k--;
            }
        }

        return ans;
    }
};
